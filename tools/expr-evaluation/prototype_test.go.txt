// The compatible prototype deliberately supports only the scalar/predicate
// subset. It does not claim to replace loops, object assembly, or the compiler.
// Real leaf evaluation is reused so Pointer, presence, precision, budgets and
// configuration locations are compared without a second implementation.
type exprEvaluationEnv struct {
	State  evaluation
	Nodes  []*compiledExpression
	Absent Value
}

func (e *exprEvaluationEnv) Leaf(id int) (Value, error) {
	return e.Nodes[id].evaluate(e.State)
}

func (e *exprEvaluationEnv) Enter(id int) (bool, error) {
	// One existing evaluator step supplies identical cancellation/budget rules.
	step := compiledExpression{op: "omit", path: e.Nodes[id].path}
	_, err := step.evaluate(e.State)
	return true, err
}

func (e *exprEvaluationEnv) Finish(id int, value Value) (Value, error) {
	if len(value.raw) > e.State.budget.limits.BufferBytes {
		return Value{}, locateMappingError(e.Nodes[id].path, resourceFailure("evaluation", "bytes", e.State.budget.limits.BufferBytes, len(value.raw)))
	}
	return value, nil
}

func (e *exprEvaluationEnv) Bool(id int, value Value) (bool, error) {
	result, err := readBoolean(value)
	return result, locateMappingError(e.Nodes[id].path, err)
}

func (e *exprEvaluationEnv) Apply(id int, values ...Value) (Value, error) {
	node := e.Nodes[id]
	fail := func(err error) (Value, error) { return Value{}, locateMappingError(node.path, err) }
	if node.op == "all" || node.op == "any" || node.op == "equal" {
		bytes := 0
		for _, value := range values {
			bytes += len(value.raw) + 1
			if bytes > e.State.budget.limits.BufferBytes {
				return fail(resourceFailure("evaluation", "bytes", e.State.budget.limits.BufferBytes, bytes))
			}
		}
	}
	var result bool
	switch node.op {
	case "exists":
		result = !values[0].IsZero()
	case "present":
		result = !values[0].IsZero() && !values[0].IsNull()
	case "equal":
		result = equalValues(values[0], values[1])
	case "not":
		v, err := readBoolean(values[0])
		if err != nil {
			return fail(err)
		}
		result = !v
	case "all", "any":
		result = node.op == "all"
		for _, value := range values {
			v, err := readBoolean(value)
			if err != nil {
				return fail(err)
			}
			if node.op == "all" {
				result = result && v
			} else {
				result = result || v
			}
		}
	default:
		return fail(fmt.Errorf("unsupported prototype operation %s", node.op))
	}
	value, err := EncodeValue(result)
	if err != nil {
		return fail(err)
	}
	return e.Finish(id, value)
}

type exprEvaluationProgram struct {
	program   *vm.Program
	nodes     []*compiledExpression
	functions bool
}

func compileExprEvaluation(root *compiledExpression, functions ...bool) (*exprEvaluationProgram, error) {
	nodes := []*compiledExpression{}
	var lower func(*compiledExpression) (string, error)
	lower = func(node *compiledExpression) (string, error) {
		id := len(nodes)
		nodes = append(nodes, node)
		switch node.op {
		case "read", "literal", "omit":
			return fmt.Sprintf("Leaf(%d)", id), nil
		case "if":
			condition, err := lower(node.when)
			if err != nil {
				return "", err
			}
			then, err := lower(node.then)
			if err != nil {
				return "", err
			}
			otherwise := "Absent"
			if node.otherwise != nil {
				otherwise, err = lower(node.otherwise)
				if err != nil {
					return "", err
				}
			}
			return fmt.Sprintf("(Enter(%d) ? Finish(%d, Bool(%d, %s) ? %s : %s) : Absent)", id, id, id, condition, then, otherwise), nil
		case "exists", "present", "not", "equal", "all", "any":
			children := node.items
			if node.source != nil {
				children = []*compiledExpression{node.source}
			}
			args := []string{fmt.Sprint(id)}
			for _, child := range children {
				source, err := lower(child)
				if err != nil {
					return "", err
				}
				args = append(args, source)
			}
			return fmt.Sprintf("(Enter(%d) ? Apply(%s) : Absent)", id, strings.Join(args, ",")), nil
		default:
			return "", fmt.Errorf("unsupported prototype operation %s", node.op)
		}
	}
	source, err := lower(root)
	if err != nil {
		return nil, err
	}
	options := []expr.Option{expr.Env(&exprEvaluationEnv{}), expr.DisableAllBuiltins(), expr.MaxNodes(uint(DefaultLimits().Nodes))}
	fast := len(functions) == 0 || functions[0]
	if fast {
		source = strings.NewReplacer("Leaf(", "leaf(E,", "Enter(", "enter(E,", "Finish(", "finish(E,", "Bool(", "boolean(E,", "Apply(", "apply(E,", "Absent", "E.Absent").Replace(source)
		options[0] = expr.Env(map[string]any{"E": &exprEvaluationEnv{}})
		options = append(options,
			expr.Function("leaf", func(p ...any) (any, error) { return p[0].(*exprEvaluationEnv).Leaf(p[1].(int)) }, new(func(*exprEvaluationEnv, int) Value)),
			expr.Function("enter", func(p ...any) (any, error) { return p[0].(*exprEvaluationEnv).Enter(p[1].(int)) }, new(func(*exprEvaluationEnv, int) bool)),
			expr.Function("finish", func(p ...any) (any, error) { return p[0].(*exprEvaluationEnv).Finish(p[1].(int), p[2].(Value)) }, new(func(*exprEvaluationEnv, int, Value) Value)),
			expr.Function("boolean", func(p ...any) (any, error) { return p[0].(*exprEvaluationEnv).Bool(p[1].(int), p[2].(Value)) }, new(func(*exprEvaluationEnv, int, Value) bool)),
			expr.Function("apply", func(p ...any) (any, error) {
				values := make([]Value, len(p)-2)
				for i := range values {
					values[i] = p[i+2].(Value)
				}
				return p[0].(*exprEvaluationEnv).Apply(p[1].(int), values...)
			}, new(func(*exprEvaluationEnv, int, ...Value) Value)),
		)
	}
	program, err := expr.Compile(source, options...)
	if err != nil {
		return nil, err
	}
	return &exprEvaluationProgram{program, nodes, fast}, nil
}

func (p *exprEvaluationProgram) run(state evaluation) (Value, error) {
	var environment any = &exprEvaluationEnv{State: state, Nodes: p.nodes}
	if p.functions {
		environment = map[string]any{"E": environment}
	}
	result, err := expr.Run(p.program, environment)
	if err != nil {
		return Value{}, err
	}
	value, ok := result.(Value)
	if !ok {
		return Value{}, fmt.Errorf("prototype result %T", result)
	}
	return value, nil
}

func exprEvaluationCompile(t testing.TB, expression Expression) *compiledExpression {
	t.Helper()
	compiler := expressionCompiler{limits: DefaultLimits()}
	compiled, err := compiler.compile(expression, "/transform", expressionScope{}, 1)
	if err != nil {
		t.Fatal(err)
	}
	return compiled
}

func exprEvaluationState(ctx context.Context, input Value, steps int) evaluation {
	return evaluation{ctx: ctx, input: input, root: input, budget: &evaluationBudget{limits: DefaultLimits(), stepLimit: steps}}
}

func exprEvaluationCases() []struct {
	name       string
	expression Expression
	input      string
} {
	read := func(path string) Expression { return Expression{Op: "read", Path: path} }
	equal := Expression{Op: "equal", Items: []Expression{read("/n"), {Op: "literal", Value: Value{raw: `9007199254740993123.0`}}}}
	exists := Expression{Op: "exists", Source: expressionPointer(read("/missing"))}
	var wide []Expression
	for range 64 {
		wide = append(wide, Expression{Op: "exists", Source: expressionPointer(read("/n"))})
	}
	return []struct {
		name       string
		expression Expression
		input      string
	}{
		{"read", read("/n"), `{"n":9007199254740993123}`},
		{"equal", equal, `{"n":9007199254740993123}`},
		{"missing", exists, `{"present":null}`},
		{"all-wide", Expression{Op: "all", Items: wide}, `{"n":null}`},
		{"conditional", Expression{Op: "if", When: &equal, Then: expressionPointer(read("/n")), Otherwise: &Expression{Op: "omit"}}, `{"n":9007199254740993123}`},
		{"large-read", read("/n"), `{"payload":"` + strings.Repeat("A", 1<<20) + `","n":1}`},
	}
}

func TestExprEvaluationCompatibleSubset(t *testing.T) {
	for _, tc := range exprEvaluationCases() {
		t.Run(tc.name, func(t *testing.T) {
			root := exprEvaluationCompile(t, tc.expression)
			program, err := compileExprEvaluation(root)
			if err != nil {
				t.Fatal(err)
			}
			input, err := ParseValue([]byte(tc.input))
			if err != nil {
				t.Fatal(err)
			}
			for _, steps := range []int{0, 1, 2, 5} {
				beforeState, afterState := exprEvaluationState(t.Context(), input, steps), exprEvaluationState(t.Context(), input, steps)
				before, beforeErr := root.evaluate(beforeState)
				after, afterErr := program.run(afterState)
				if before.raw != after.raw || (beforeErr == nil) != (afterErr == nil) || beforeState.budget.steps != afterState.budget.steps {
					t.Fatalf("steps=%d before=%q/%v/%d after=%q/%v/%d", steps, before.raw, beforeErr, beforeState.budget.steps, after.raw, afterErr, afterState.budget.steps)
				}
				if beforeErr != nil && mappingErrorPath("", beforeErr) != mappingErrorPath("", afterErr) {
					t.Fatalf("location lost: %v / %v", beforeErr, afterErr)
				}
			}
			ctx, cancel := context.WithCancel(t.Context())
			cancel()
			_, err = program.run(exprEvaluationState(ctx, input, 0))
			if !errors.Is(err, context.Canceled) {
				t.Fatalf("cancellation cause lost: %v", err)
			}
			for range 4 {
				t.Run("concurrent", func(t *testing.T) {
					t.Parallel()
					_, err := program.run(exprEvaluationState(t.Context(), input, 0))
					if err != nil {
						t.Fatal(err)
					}
				})
			}
		})
	}
}

func TestExprEvaluationNativeDifferences(t *testing.T) {
	run := func(source string, env map[string]any, options ...expr.Option) any {
		t.Helper()
		options = append([]expr.Option{expr.Env(env)}, options...)
		program, err := expr.Compile(source, options...)
		if err != nil {
			t.Fatal(err)
		}
		result, err := expr.Run(program, env)
		if err != nil {
			t.Fatal(err)
		}
		return result
	}
	if got := run("a == b", map[string]any{"a": json.Number("1"), "b": json.Number("1.0")}); got != false {
		t.Fatalf("unexpected Number equality: %v", got)
	}
	if got := run("a == b", map[string]any{"a": float64(9007199254740992), "b": float64(9007199254740993)}); got != true {
		t.Fatalf("unexpected float equality: %v", got)
	}
	for _, input := range []map[string]any{{}, {"x": nil}} {
		if got := run("input.x == nil", map[string]any{"input": input}); got != true {
			t.Fatalf("native missing/null: %v", got)
		}
	}
	calls := 0
	if got := run("flag && fail()", map[string]any{"flag": false, "fail": func() bool { calls++; return true }}); got != false || calls != 0 {
		t.Fatal("expected native short circuit")
	}
	ctx, cancel := context.WithCancel(t.Context())
	cancel()
	if got := run("len(map(1..32, {# * 2}))", map[string]any{"ctx": ctx}, expr.WithContext("ctx")); got != 32 {
		t.Fatalf("native cancellation probe: %v", got)
	}
	program, err := expr.Compile("1..32")
	if err != nil {
		t.Fatal(err)
	}
	machine := vm.VM{MemoryBudget: 8}
	if _, err := machine.Run(program, nil); err == nil {
		t.Fatal("expected native range memory limit")
	}
	program, err = expr.Compile("payload", expr.Env(map[string]any{"payload": ""}))
	if err != nil {
		t.Fatal(err)
	}
	if result, err := machine.Run(program, map[string]any{"payload": strings.Repeat("x", 1024)}); err != nil || len(result.(string)) != 1024 {
		t.Fatalf("memory budget is not a string-byte budget: %v", err)
	}
}

func TestExprEvaluationEagerFailureAndBytes(t *testing.T) {
	root := exprEvaluationCompile(t, Expression{Op: "all", Items: []Expression{literalExpression(false), {Op: "read", Path: "/missing", Required: true}}})
	program, err := compileExprEvaluation(root)
	if err != nil {
		t.Fatal(err)
	}
	input := Value{raw: `{}`}
	_, before := root.evaluate(exprEvaluationState(t.Context(), input, 0))
	_, after := program.run(exprEvaluationState(t.Context(), input, 0))
	if before == nil || after == nil || mappingErrorPath("", before) != mappingErrorPath("", after) {
		t.Fatalf("eager required failure mismatch: %v / %v", before, after)
	}
	root = exprEvaluationCompile(t, Expression{Op: "equal", Items: []Expression{literalExpression("long"), literalExpression("long")}})
	program, err = compileExprEvaluation(root)
	if err != nil {
		t.Fatal(err)
	}
	for _, run := range []func(evaluation) (Value, error){root.evaluate, program.run} {
		state := exprEvaluationState(t.Context(), input, 0)
		state.budget.limits.BufferBytes = 8
		if _, err := run(state); err == nil {
			t.Fatal("operand byte budget bypassed")
		}
	}
}

// This remaining mismatch is an adoption blocker, not an accepted behavior
// change: aggregate callbacks run only after all arguments have been evaluated.
func TestExprEvaluationOperandBudgetOrderingGap(t *testing.T) {
	root := exprEvaluationCompile(t, Expression{Op: "all", Items: []Expression{
		literalExpression(true), literalExpression(true), {Op: "read", Path: "/missing", Required: true},
	}})
	program, err := compileExprEvaluation(root)
	if err != nil {
		t.Fatal(err)
	}
	beforeState, afterState := exprEvaluationState(t.Context(), Value{raw: `{}`}, 0), exprEvaluationState(t.Context(), Value{raw: `{}`}, 0)
	beforeState.budget.limits.BufferBytes = 8
	afterState.budget.limits.BufferBytes = 8
	_, before := root.evaluate(beforeState)
	_, after := program.run(afterState)
	if before == nil || after == nil || beforeState.budget.steps != 3 || afterState.budget.steps != 4 || !strings.Contains(after.Error(), "required field is absent") {
		t.Fatalf("expected documented ordering gap: %v (%d) / %v (%d)", before, beforeState.budget.steps, after, afterState.budget.steps)
	}
}

type exprEvaluationCancelContext struct {
	context.Context
	cancel context.CancelFunc
	checks int
}

func (c *exprEvaluationCancelContext) Err() error {
	c.checks++
	if c.checks == 4 {
		c.cancel()
	}
	return c.Context.Err()
}

func TestExprEvaluationCancellationBetweenNodes(t *testing.T) {
	root := exprEvaluationCompile(t, exprEvaluationCases()[3].expression)
	program, err := compileExprEvaluation(root)
	if err != nil {
		t.Fatal(err)
	}
	var paths []string
	for _, run := range []func(evaluation) (Value, error){root.evaluate, program.run} {
		base, cancel := context.WithCancel(t.Context())
		ctx := &exprEvaluationCancelContext{Context: base, cancel: cancel}
		state := exprEvaluationState(ctx, Value{raw: `{"n":null}`}, 0)
		_, err := run(state)
		cancel()
		if !errors.Is(err, context.Canceled) || state.budget.steps != 4 {
			t.Fatalf("mid-evaluation cancellation: %v steps=%d", err, state.budget.steps)
		}
		paths = append(paths, mappingErrorPath("", err))
	}
	if paths[0] != paths[1] {
		t.Fatalf("cancellation paths: %v", paths)
	}
}

func FuzzExprEvaluationSubset(f *testing.F) {
	var roots []*compiledExpression
	var programs []*exprEvaluationProgram
	for _, tc := range exprEvaluationCases() {
		if tc.name == "large-read" {
			continue
		}
		root := exprEvaluationCompile(f, tc.expression)
		program, err := compileExprEvaluation(root)
		if err != nil {
			f.Fatal(err)
		}
		roots = append(roots, root)
		programs = append(programs, program)
	}
	for _, raw := range []string{`{}`, `null`, `false`, `{"n":null}`, `{"n":false}`, `{"n":0}`, `{"n":""}`, `{"n":[]}`, `{"n":9007199254740993123}`, `{"n":9007199254740993123.0}`, `{"n":1,"n":2}`, `{"n":{"a":1}}`} {
		f.Add(raw, uint8(0))
	}
	f.Fuzz(func(t *testing.T, raw string, steps uint8) {
		if len(raw) > 16384 {
			t.Skip()
		}
		input, err := ParseValue([]byte(raw))
		if err != nil {
			t.Skip()
		}
		if err := checkValueLimits(input, DefaultLimits()); err != nil {
			t.Skip()
		}
		for i, root := range roots {
			beforeState, afterState := exprEvaluationState(t.Context(), input, int(steps)), exprEvaluationState(t.Context(), input, int(steps))
			before, beforeErr := root.evaluate(beforeState)
			after, afterErr := programs[i].run(afterState)
			if before.raw != after.raw || (beforeErr == nil) != (afterErr == nil) || beforeState.budget.steps != afterState.budget.steps || mappingErrorPath("", beforeErr) != mappingErrorPath("", afterErr) {
				t.Fatalf("case %d input %q: before=%q/%v/%d after=%q/%v/%d", i, raw, before.raw, beforeErr, beforeState.budget.steps, after.raw, afterErr, afterState.budget.steps)
			}
		}
	})
}

func BenchmarkExprEvaluation(b *testing.B) {
	for _, tc := range exprEvaluationCases() {
		root := exprEvaluationCompile(b, tc.expression)
		program, err := compileExprEvaluation(root)
		if err != nil {
			b.Fatal(err)
		}
		methods, err := compileExprEvaluation(root, false)
		if err != nil {
			b.Fatal(err)
		}
		input, err := ParseValue([]byte(tc.input))
		if err != nil {
			b.Fatal(err)
		}
		for _, candidate := range []struct {
			name string
			run  func(evaluation) (Value, error)
		}{{"current", root.evaluate}, {"expr-functions", program.run}, {"expr-methods", methods.run}} {
			b.Run(tc.name+"/"+candidate.name, func(b *testing.B) {
				b.ReportAllocs()
				for b.Loop() {
					_, err := candidate.run(exprEvaluationState(context.Background(), input, 0))
					if err != nil {
						b.Fatal(err)
					}
				}
			})
		}
	}
}
